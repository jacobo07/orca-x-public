<objective>
Ship the shared availability model, the capability set, and the host producer that carries capabilities additively
in `status.get`, so a phone can compose what Orca knows about a host and what the host can do.

Purpose: success criteria 1 and 2 of phase 18. Power, orca and connectivity are three composed fields (D-01). No
response means unknown, never powered-off (D-02). Orca derives from fields the host already reports (D-03).
Capabilities have values and reasons and stay apart from state and from grants (D-04). The host reports them
behind a new runtime capability string, filled from phase 17 producers (D-05). The change is additive and adds no
stream opcode (D-07).
Output: `src/shared/host-availability.ts`, the host producer and autostart probe, the `status.get` field, and the
index.ts wiring, each with tests. Node-side single vitest files run on this laptop (D-11).
</objective>

New, `src/shared/host-availability.ts` (imports only `HOST_AVAILABILITY_RUNTIME_CAPABILITY` from `./protocol-version`; no Node types, because mobile imports this file):

```ts
export const HOST_EVIDENCE_KINDS = ['live-response', 'transport-observed', 'host-announced', 'relay-reported', 'os-probe'] as const
export type HostEvidenceKind = (typeof HOST_EVIDENCE_KINDS)[number]
export type HostEvidence = { kind: HostEvidenceKind; observedAt: number }

export const HOST_POWER_STATES = ['awake', 'sleeping', 'hibernating', 'powered-off', 'unknown', 'unobservable'] as const
export const HOST_ORCA_STATES = ['service-ready', 'runtime-ready', 'ui-open', 'ui-closed', 'starting', 'version-mismatch', 'unavailable', 'unknown'] as const
export const HOST_CONNECTIVITY_STATES = ['normal', 'warning', 'unreachable', 'auth-failed', 'wake-proxy-available', 'oob-available'] as const
export type HostPowerState / HostOrcaState / HostConnectivityState  // (typeof X)[number]

export type HostObserved<S extends string> =
  | { value: Exclude<S, 'unknown'>; evidence: HostEvidence; detail?: string }
  | { value: 'unknown'; evidence: null; reason: string }
export type HostAvailability = {
  power: HostObserved<HostPowerState>
  orca: HostObserved<HostOrcaState>
  connectivity: { value: HostConnectivityState; evidence: HostEvidence }
}
export type HostProcessKind = 'serve' | 'desktop'
export type HostStatusObservation = {
  observedAt: number
  protocolCompatible: boolean
  graphStatus?: string
  desktopWindowStatus?: string
  processKind?: HostProcessKind | null
}
export type HostAvailabilityObservations = {
  transport: { verdictKind: 'normal' | 'warning' | 'unreachable' | 'auth-failed'; connected: boolean; observedAt: number }
  status: HostStatusObservation | null   // latest status.get answer on the CURRENT connection, else null
}
export function composeHostAvailability(observations: HostAvailabilityObservations): HostAvailability

export const HOST_CAPABILITY_IDS = ['can-start-orca', 'can-wake-lan', 'can-wake-remote', 'can-power-on-oob', 'can-relay', 'can-direct'] as const
export const HOST_CAPABILITY_VALUES = ['supported', 'not-supported', 'not-configured', 'needs-owner-setup', 'unknown'] as const
export type HostCapabilityId / HostCapabilityValue
export type HostCapability =
  | { value: Exclude<HostCapabilityValue, 'unknown'>; reason: string; evidence: HostEvidence }
  | { value: 'unknown'; reason: string; evidence: null }
export type HostCapabilitySet = Record<HostCapabilityId, HostCapability>
export type HostAvailabilityAnnouncement = { schema: 1; processKind: HostProcessKind | null; observedAt: number; capabilities: HostCapabilitySet }
export function unknownHostCapabilities(reason: string): HostCapabilitySet
export function parseHostAvailabilityAnnouncement(status: unknown): { processKind: HostProcessKind | null; capabilities: HostCapabilitySet }
```
